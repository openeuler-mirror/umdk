# case 12: stress conservation (C8)
# 8 workers, 10000 iterations of: alloc 10 sc0 bufs (parallel) then free all
# (parallel). The longest-running test in the suite. Verifies:
#   - Long-running conservation: per-sc global+tls+alloced stays constant
#     across 10000 iterations of churn
#   - TLS self_shrink path exercised thousands of times (each iter's alloc
#     triggers self_shrink on the TLS surplus from prior free)
#   - No accumulated leaks (alloced returns to 0 at iter boundary)
#   - DFX counters (AccFetchCnt/AccAllocCnt/AccFreeCnt) grow proportionally
#
# Time budget: ~1-3 seconds on a typical 8-core box. Watch CPU usage of
# the 8 worker processes — they should pin at ~100% per core during the
# repeat block (spinlock contention is hot).
#
# Note: do NOT insert status inside the repeat block (forbidden by the
# parser; conservation must be checked after repeat completes).
init count=2 blockSizes=4K,64K mode=split scaleCap=on totalSz=200M threads=8

# Bootstrap each worker's TLS so the first repeat iter has self_shrink
# to perform (otherwise iter 1 just does cold fetch_from_global).
thread_0 alloc 4096 100
thread_0 free all
thread_1 alloc 4096 100
thread_1 free all
thread_2 alloc 4096 100
thread_2 free all
thread_3 alloc 4096 100
thread_3 free all
thread_4 alloc 4096 100
thread_4 free all
thread_5 alloc 4096 100
thread_5 free all
thread_6 alloc 4096 100
thread_6 free all
thread_7 alloc 4096 100
thread_7 free all

# 10000 iterations: alloc 10 + free all, 8 workers parallel each phase
repeat 10000
  parallel
    thread_0 alloc 4096 10
    thread_1 alloc 4096 10
    thread_2 alloc 4096 10
    thread_3 alloc 4096 10
    thread_4 alloc 4096 10
    thread_5 alloc 4096 10
    thread_6 alloc 4096 10
    thread_7 alloc 4096 10
  join
  parallel
    thread_0 free all
    thread_1 free all
    thread_2 free all
    thread_3 free all
    thread_4 free all
    thread_5 free all
    thread_6 free all
    thread_7 free all
  join
end

# After 10000 iterations:
#   - alloced sc0 should be 0 (every iter balanced alloc+free)
#   - TLS rows should show stable fetch bytes (self_shrink returned excess)
#   - global_sc0 should be near initial (some TLS surplus may persist)
#   - DFX: AccFetchCnt, AccAllocCnt, AccFreeCnt should be in expected ratio
status
