# case 23: stress_repeat infinite with `free <idx>` mode
#
# Verifies Worker::alloc_ops is cleared at iter boundary even when user
# uses `free <idx>` (which doesn't auto-clear alloc_ops like `free all`).
# If alloc_ops weren't cleared, it would grow unboundedly across iters
# in an infinite loop → guaranteed OOM within minutes.
#
# INFINITE — Ctrl-C to stop. Let it run for at least 1 minute to be sure
# memory is stable, then check `status` after Ctrl-C.
init count=2 blockSizes=4K,64K mode=split scaleCap=on totalSz=200M threads=2

# 2 workers, each does alloc 1 + free idx 0 per iter.
# Worker::alloc_ops must be cleared at iter boundary by DoStressRepeatBlock.
stress_repeat
  thread_0 alloc 4096 1
  thread_0 free 0
  thread_1 alloc 4096 1
  thread_1 free 0
  snapshot 1000
  iterations 200
end

status
